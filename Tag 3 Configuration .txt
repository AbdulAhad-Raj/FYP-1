
#include <WiFi.h>

// =====================================
// SENTINEL-X: TAG-03 CONFIGURATION
// =====================================

const char* TAG_ID = "TAG-03";
const char* TAG_SSID = "SX_TAG_03";
const char* TAG_PASSWORD = "SentinelTag26";

const int WIFI_CHANNEL = 6;
const int MAX_CONNECTIONS = 4;

void setup() {

  Serial.begin(115200);
  delay(1000);

  Serial.println();
  Serial.println("================================");
  Serial.println("       SENTINEL-X TAG-03");
  Serial.println("================================");

  WiFi.mode(WIFI_AP);

  Serial.println("Starting Wi-Fi Access Point...");

  bool result = WiFi.softAP(
    TAG_SSID,
    TAG_PASSWORD,
    WIFI_CHANNEL,
    false,
    MAX_CONNECTIONS
  );

  if (result) {

    Serial.println();
    Serial.println("ACCESS POINT STARTED SUCCESSFULLY");

    Serial.print("Tag ID: ");
    Serial.println(TAG_ID);

    Serial.print("Wi-Fi SSID: ");
    Serial.println(TAG_SSID);

    Serial.print("IP Address: ");
    Serial.println(WiFi.softAPIP());

    Serial.print("Tag MAC Address: ");
    Serial.println(WiFi.softAPmacAddress());

    Serial.print("Wi-Fi Channel: ");
    Serial.println(WIFI_CHANNEL);

    Serial.println();
    Serial.println("TAG-03 IS READY FOR DETECTION");

  }
  else {

    Serial.println("ERROR: ACCESS POINT FAILED TO START");

  }
}

void loop() {

  delay(1000);

}
