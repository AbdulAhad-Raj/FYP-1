
#include <WiFi.h>

// ============================================
// SENTINEL-X MODULE 2
// STRUCTURED RSSI DATA COLLECTION
// ============================================

// Anchor identification
const char* ANCHOR_ID = "ANC-01";

// Tracking tag SSIDs
const char* TAG_01 = "SX_TAG_01";
const char* TAG_02 = "SX_TAG_02";
const char* TAG_03 = "SX_TAG_03";
const char* TAG_04 = "SX_TAG_04";

// Scan interval (milliseconds)
const unsigned long SCAN_INTERVAL = 3000;

unsigned long previousScanTime = 0;

// ============================================
// SETUP
// ============================================

void setup() {

  Serial.begin(115200);
  delay(1000);

  Serial.println();
  Serial.println("SENTINEL-X MODULE 2");
  Serial.println("RSSI DATA COLLECTION");

  Serial.print("Anchor: ");
  Serial.println(ANCHOR_ID);

  // Configure ESP32 as Wi-Fi station
  WiFi.mode(WIFI_STA);

  WiFi.disconnect();

  delay(500);

  // CSV column headings
  Serial.println();
  Serial.println("ANCHOR_ID,TAG_ID,RSSI_DBM,CHANNEL,BSSID,ELAPSED_MS");
}

// ============================================
// PRINT STRUCTURED MEASUREMENT
// ============================================

void printMeasurement(
  int networkIndex,
  const char* tagID
) {

  Serial.print(ANCHOR_ID);
  Serial.print(",");

  Serial.print(tagID);
  Serial.print(",");

  Serial.print(WiFi.RSSI(networkIndex));
  Serial.print(",");

  Serial.print(WiFi.channel(networkIndex));
  Serial.print(",");

  Serial.print(WiFi.BSSIDstr(networkIndex));
  Serial.print(",");

  Serial.println(millis());
}

// ============================================
// MAIN LOOP
// ============================================

void loop() {

  unsigned long currentTime = millis();

  if (currentTime - previousScanTime >= SCAN_INTERVAL) {

    previousScanTime = currentTime;

    // Scan available Wi-Fi networks
    int networkCount = WiFi.scanNetworks(false, true);

    if (networkCount >= 0) {

      for (int i = 0; i < networkCount; i++) {

        String detectedSSID = WiFi.SSID(i);

        if (detectedSSID == TAG_01) {
          printMeasurement(i, "TAG-01");
        }

        else if (detectedSSID == TAG_02) {
          printMeasurement(i, "TAG-02");
        }

        else if (detectedSSID == TAG_03) {
          printMeasurement(i, "TAG-03");
        }

        else if (detectedSSID == TAG_04) {
          printMeasurement(i, "TAG-04");
        }
      }
    }
    else {
      Serial.println("SCAN_ERROR");
    }

    // Release previous scan results
    WiFi.scanDelete();
  }
}
